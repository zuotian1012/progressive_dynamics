cmake_minimum_required(VERSION 3.20)
project(pd_intrinsic_builder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SSP_ROOT "${PROJECT_ROOT}/third_party/surface_multigrid" CACHE PATH "Author's intrinsic prolongation implementation")
set(EIGEN_ROOT "${PROJECT_ROOT}/third_party/eigen" CACHE PATH "Eigen header root")
set(IGL_ROOT "${PROJECT_ROOT}/third_party/libigl" CACHE PATH "Pinned libigl source root")
file(GLOB SSP_SOURCES "${SSP_ROOT}/src/*.cpp")
list(FILTER SSP_SOURCES EXCLUDE REGEX "/(mg_|min_quad_with_fixed_mg|SSP_random|query_coarse_to_fine|normalize_unit_area|get_prolong)")
list(FILTER SSP_SOURCES EXCLUDE REGEX "/SSP_qslim.cpp$")
add_executable(intrinsic_builder main.cpp SSP_qslim_probabilistic.cpp ${SSP_SOURCES})
target_include_directories(intrinsic_builder PRIVATE "${PROJECT_ROOT}/third_party/probabilistic_quadrics")
target_include_directories(intrinsic_builder PRIVATE "${SSP_ROOT}/src" "${IGL_ROOT}/include" "${EIGEN_ROOT}")
target_compile_definitions(intrinsic_builder PRIVATE NOMINMAX _USE_MATH_DEFINES _CRT_SECURE_NO_WARNINGS)
if(MSVC)
  target_compile_options(intrinsic_builder PRIVATE /bigobj /EHsc /utf-8)
endif()
